🚀 QUICK START (Copy-Paste These Commands)

ON SERVER COMPUTER:
-------------------

# 1. Make scripts executable
chmod +x scripts/*.sh

# 2. Run setup (one-time)
./scripts/setup-network.sh

# 3. Setup database (one-time)
sudo mysql -u root -p

CREATE DATABASE attendance_system;
CREATE USER 'attendance_user'@'%' IDENTIFIED BY 'SecurePassword123!';
GRANT ALL PRIVILEGES ON attendance_system.* TO 'attendance_user'@'%';
FLUSH PRIVILEGES;
EXIT;

mysql -u attendance_user -p attendance_system < scripts/database/setup-database.sql

# 4. Start server
./scripts/start-server-network.sh

# ✅ Server will show: "Server IP: 192.168.1.100" (write this down!)


ON CLIENT COMPUTER:
-------------------

# 1. Make scripts executable
chmod +x scripts/*.sh

# 2. Run setup (one-time)
./scripts/setup-network.sh

# 3. Start client (replace with YOUR server IP)
./scripts/start-client-network.sh 192.168.1.100

# ✅ Login screen appears!
# Username: admin
# Password: Admin@123


================================================================================
📋 DETAILED COMMANDS
================================================================================

FIND SERVER IP:
---------------
hostname -I                    # Linux/Mac
ipconfig                       # Windows
ip addr show                   # Linux alternative


TEST CONNECTION:
----------------
ping 192.168.1.100            # Test if server is reachable
telnet 192.168.1.100 1099     # Test if RMI port is open
nc -zv 192.168.1.100 1099     # Alternative port test


CHECK SERVER STATUS:
--------------------
netstat -tulpn | grep 1099    # Check if port is listening
ps aux | grep ServerLauncher  # Check if server is running
tail -f logs/server.log       # View server logs


FIREWALL CONFIGURATION:
-----------------------
# Ubuntu/Debian
sudo ufw allow 1099/tcp
sudo ufw status

# CentOS/RHEL
sudo firewall-cmd --permanent --add-port=1099/tcp
sudo firewall-cmd --reload


MYSQL COMMANDS:
---------------
sudo systemctl start mysql     # Start MySQL
sudo systemctl status mysql    # Check MySQL status
mysql -u root -p              # Login to MySQL


BUILD PROJECT:
--------------
mvn clean package -DskipTests  # Build JAR file


STOP SERVER/CLIENT:
-------------------
Ctrl+C                        # In terminal
kill <PID>                    # Kill process


================================================================================
🌐 NETWORK SCENARIOS
================================================================================

SAME WIFI (EASIEST):
--------------------
# Both computers on same WiFi
# Server: ./scripts/start-server-network.sh
# Client: ./scripts/start-client-network.sh 192.168.1.100


ETHERNET CABLE:
---------------
# Server:
sudo ip addr add 192.168.100.1/24 dev eth0
./scripts/start-server-network.sh

# Client:
sudo ip addr add 192.168.100.2/24 dev eth0
./scripts/start-client-network.sh 192.168.100.1


MOBILE HOTSPOT:
---------------
# Server creates hotspot:
nmcli device wifi hotspot ssid "AttendanceServer" password "password123"
./scripts/start-server-network.sh

# Client connects to hotspot:
./scripts/start-client-network.sh 192.168.137.1


VPN (ZEROTIER):
---------------
# Both computers:
curl -s https://install.zerotier.com | sudo bash
sudo zerotier-cli join <network-id>

# Server:
./scripts/start-server-network.sh

# Client:
./scripts/start-client-network.sh 10.147.17.10


================================================================================
🔥 TROUBLESHOOTING
================================================================================

PROBLEM: Connection refused
SOLUTION:
  netstat -tulpn | grep 1099
  sudo ufw allow 1099/tcp
  ./scripts/start-server-network.sh


PROBLEM: Cannot reach server
SOLUTION:
  ping 192.168.1.100
  hostname -I  # on server to verify IP


PROBLEM: Port already in use
SOLUTION:
  sudo netstat -tulpn | grep 1099
  kill <PID>


PROBLEM: JAR file not found
SOLUTION:
  mvn clean package -DskipTests


PROBLEM: MySQL not running
SOLUTION:
  sudo systemctl start mysql
  sudo systemctl enable mysql
